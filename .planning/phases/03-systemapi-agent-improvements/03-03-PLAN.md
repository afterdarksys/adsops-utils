Add a namespace validation helper (T-03-09 mitigation):
```go
// isValidNamespace validates that a namespace string is safe for URL construction.
// Rejects values containing path separators, traversal sequences, or non-alphanumeric chars
// beyond dash and dot (matching Kubernetes namespace naming rules).
func isValidNamespace(ns string) bool {
    if len(ns) == 0 || len(ns) > 253 {
        return false
    }
    for _, r := range ns {
        if !((r >= 'a' && r <= 'z') || (r >= '0' && r <= '9') || r == '-') {
            return false
        }
    }
    return true
}
```

Helper for resource path mapping:
```go
func resourcePath(apiVersion, kind, namespace string) (string, error) {
    kind = strings.ToLower(kind) + "s" // simple pluralization
    if namespace == "" {
        namespace = "default"
    }
    if !isValidNamespace(namespace) {
        return "", fmt.Errorf("invalid namespace: %q", namespace)
    }
    switch {
    case apiVersion == "v1":
        return fmt.Sprintf("/api/v1/namespaces/%s/%s", namespace, kind), nil
    case strings.HasPrefix(apiVersion, "apps/"):
        return fmt.Sprintf("/apis/%s/namespaces/%s/%s", apiVersion, namespace, kind), nil
    default:
        return fmt.Sprintf("/apis/%s/namespaces/%s/%s", apiVersion, namespace, kind), nil
    }
}
```
