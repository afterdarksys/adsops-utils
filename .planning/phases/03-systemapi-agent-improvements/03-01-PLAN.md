const dockerAPIVersion = "v1.41"

// DockerClient communicates with the Docker Engine API over a Unix socket.
type DockerClient struct {
    client     *http.Client
    socketPath string
}

// DockerContainer represents a container from the Docker API list endpoint.
type DockerContainer struct {
    Id    string `json:"Id"`
    Names []string `json:"Names"`
    Image string `json:"Image"`
    State string `json:"State"`
}

// NewDockerClient creates a client that talks to the Docker Engine API.
// Per D-01: reads DOCKER_HOST env var; falls back to /var/run/docker.sock.
func NewDockerClient() *DockerClient {
    socketPath := "/var/run/docker.sock"
    if h := os.Getenv("DOCKER_HOST"); h != "" {
        if strings.HasPrefix(h, "unix://") {
            socketPath = strings.TrimPrefix(h, "unix://")
        } else {
            socketPath = h
        }
    }
    transport := &http.Transport{
        DialContext: func(ctx context.Context, _, _ string) (net.Conn, error) {
            return (&net.Dialer{}).DialContext(ctx, "unix", socketPath)
        },
    }
    return &DockerClient{
        client:     &http.Client{Transport: transport, Timeout: 10 * time.Second},
        socketPath: socketPath,
    }
}

package main

import (
    "crypto/tls"
    "crypto/x509"
    "encoding/base64"
    "fmt"
    "net/http"
    "os"
    "time"

    "gopkg.in/yaml.v3"
)

// K3sClient communicates with the Kubernetes API using TLS from kubeconfig.
type K3sClient struct {
    client    *http.Client
    serverURL string
}

// kubeconfig represents the minimal kubeconfig YAML structure.
type kubeconfig struct {
    Clusters []struct {
        Cluster struct {
            Server                   string `yaml:"server"`
            CertificateAuthorityData string `yaml:"certificate-authority-data"`
        } `yaml:"cluster"`
    } `yaml:"clusters"`
    Users []struct {
        User struct {
            ClientCertificateData string `yaml:"client-certificate-data"`
            ClientKeyData         string `yaml:"client-key-data"`
        } `yaml:"user"`
    } `yaml:"users"`
}

// NewK3sClient creates a client from kubeconfig.
// Per D-04: tries /etc/rancher/k3s/k3s.yaml first, then KUBECONFIG env var.
func NewK3sClient() (*K3sClient, error) {
    kubeconfigPath := "/etc/rancher/k3s/k3s.yaml"
    if _, err := os.Stat(kubeconfigPath); os.IsNotExist(err) {
        kubeconfigPath = os.Getenv("KUBECONFIG")
        if kubeconfigPath == "" {
            return nil, fmt.Errorf("no kubeconfig found: /etc/rancher/k3s/k3s.yaml missing and KUBECONFIG not set")
        }
    }

    data, err := os.ReadFile(kubeconfigPath)
    if err != nil {
        return nil, fmt.Errorf("read kubeconfig: %w", err)
    }

    var kc kubeconfig
    if err := yaml.Unmarshal(data, &kc); err != nil {
        return nil, fmt.Errorf("parse kubeconfig: %w", err)
    }
    if len(kc.Clusters) == 0 || len(kc.Users) == 0 {
        return nil, fmt.Errorf("kubeconfig missing clusters or users")
    }

    // Decode base64-encoded PEM certs (Pitfall 3 from RESEARCH.md)
    caCertPEM, err := base64.StdEncoding.DecodeString(kc.Clusters[0].Cluster.CertificateAuthorityData)
    if err != nil {
        return nil, fmt.Errorf("decode CA cert: %w", err)
    }
    clientCertPEM, err := base64.StdEncoding.DecodeString(kc.Users[0].User.ClientCertificateData)
    if err != nil {
        return nil, fmt.Errorf("decode client cert: %w", err)
    }
    clientKeyPEM, err := base64.StdEncoding.DecodeString(kc.Users[0].User.ClientKeyData)
    if err != nil {
        return nil, fmt.Errorf("decode client key: %w", err)
    }

    caCertPool := x509.NewCertPool()
    caCertPool.AppendCertsFromPEM(caCertPEM)
    tlsCert, err := tls.X509KeyPair(clientCertPEM, clientKeyPEM)
    if err != nil {
        return nil, fmt.Errorf("load TLS keypair: %w", err)
    }

    transport := &http.Transport{
        TLSClientConfig: &tls.Config{
            RootCAs:      caCertPool,
            Certificates: []tls.Certificate{tlsCert},
        },
    }

    return &K3sClient{
        client:    &http.Client{Transport: transport, Timeout: 10 * time.Second},
        serverURL: kc.Clusters[0].Cluster.Server,
    }, nil
}

**5. Add Thread.Load to Execute()** (lines 42-63). Insert the load function and cache BEFORE the `thread := &starlark.Thread{` declaration, and add `Load: loadFn` to the thread. Include path traversal protection (T-03-01 mitigation):
```go
func (engine *SysscriptEngine) Execute(scriptSource string) (string, error) {
    var outBuffer bytes.Buffer

    const scriptsBase = "/etc/systemapi/scripts"
    cache := make(map[string]starlark.StringDict)
    var loadFn func(*starlark.Thread, string) (starlark.StringDict, error)
    loadFn = func(thread *starlark.Thread, module string) (starlark.StringDict, error) {
        if g, ok := cache[module]; ok {
            return g, nil
        }
        // T-03-01: path traversal protection
        resolved := filepath.Clean(filepath.Join(scriptsBase, module))
        if !strings.HasPrefix(resolved, scriptsBase+string(os.PathSeparator)) && resolved != scriptsBase {
            return nil, fmt.Errorf("load(%q): path traversal not allowed", module)
        }
        src, err := os.ReadFile(resolved)
        if err != nil {
            return nil, fmt.Errorf("load(%q): %w", module, err) // D-09: fail fast
        }
        mt := &starlark.Thread{Name: "load " + module, Load: loadFn} // Pitfall 5: sub-thread needs Load too
        g, err := starlark.ExecFile(mt, module, src, nil)
        if err != nil {
            return nil, err
        }
        cache[module] = g
        return g, nil
    }

    thread := &starlark.Thread{
        Name:  "Sysscript",
        Print: func(_ *starlark.Thread, msg string) { outBuffer.WriteString(msg + "\n") },
        Load:  loadFn,
    }

    sysModule := engine.buildSysModule()
    predeclared := starlark.StringDict{"sys": sysModule}
    _, err := starlark.ExecFile(thread, "sysscript.star", scriptSource, predeclared)
    if err != nil {
        return outBuffer.String(), err
    }
    return outBuffer.String(), nil
}
```
