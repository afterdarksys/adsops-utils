```python
import asyncio
import asyncssh
import os

async def run_command(host: str, command: str) -> tuple[str, str]:
    """Run command on remote host via SSH agent (D-05)."""
    if not os.environ.get("SSH_AUTH_SOCK"):
        raise RuntimeError("No SSH agent found. Run: eval $(ssh-agent) && ssh-add")
    async with asyncssh.connect(host, known_hosts=None) as conn:
        result = await conn.run(command, check=False)
        return result.stdout or "", result.stderr or ""

async def stream_command(host: str, command: str) -> None:
    """Stream command output line-by-line (for logs -f, exec)."""
    if not os.environ.get("SSH_AUTH_SOCK"):
        raise RuntimeError("No SSH agent found. Run: eval $(ssh-agent) && ssh-add")
    async with asyncssh.connect(host, known_hosts=None) as conn:
        async with conn.create_process(command) as proc:
            async for line in proc.stdout:
                print(f"[{host}] {line}", end="")
            async for line in proc.stderr:
                print(f"[{host}] ERR: {line}", end="", file=__import__('sys').stderr)

async def run_parallel(hosts: list[str], command: str) -> list[tuple[str, str, str]]:
    """Run same command on multiple hosts (D-07)."""
    async def _run(h: str) -> tuple[str, str, str]:
        stdout, stderr = await run_command(h, command)
        return h, stdout, stderr
    return await asyncio.gather(*[_run(h) for h in hosts], return_exceptions=True)

def run_sync(host: str, command: str) -> tuple[str, str]:
    """Sync wrapper for Typer commands (D-06)."""
    return asyncio.run(run_command(host, command))

def run_parallel_sync(hosts: list[str], command: str) -> list[tuple[str, str, str]]:
    """Sync multi-host wrapper (D-06 + D-07)."""
    return asyncio.run(run_parallel(hosts, command))
```

app.add_typer(k3s_app, name="k3s")

@docker_app.command("ls")
def docker_ls_cmd(
    hosts: List[str] = typer.Argument(..., help="Host(s) to query"),
    all_: Annotated[bool, typer.Option("--all", "-a")] = False,
):
    """List Docker containers on remote host(s) (D-07: multi-host)."""
    from adsops.infractl.ssh import run_parallel_sync
    fmt = "table {{.ID}}\\t{{.Names}}\\t{{.Image}}\\t{{.Status}}\\t{{.Ports}}"
    flag = "--all " if all_ else ""
    results = run_parallel_sync(hosts, f"docker ps {flag}--format '{fmt}'")
    for item in results:
        if isinstance(item, Exception):
            typer.echo(f"ERROR: {item}", err=True)
        else:
            host, stdout, stderr = item
            if len(hosts) > 1:
                for line in stdout.splitlines():
                    typer.echo(f"[{host}] {line}")
            else:
                typer.echo(stdout, nl=False)
            if stderr:
                typer.echo(stderr, err=True)

# Similar pattern for start, stop, restart (multi-host, list of containers)
# logs and exec are single-host streaming commands

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| CLI args -> SSH command | User-supplied hostname and container names composed into SSH commands |
| SSH -> remote host | Commands executed on remote systems via asyncssh |
| Local file -> remote host | k3s apply SCPs local YAML to remote /tmp |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-05 | Tampering | infractl/docker.py | mitigate | Container names and hosts are passed as arguments to SSH commands, not shell-interpolated; asyncssh conn.run() does not use shell expansion when command is a simple string. Validate container names contain only [a-zA-Z0-9_.-] |
| T-02-06 | Tampering | infractl/k3s.py (apply) | mitigate | SCP file path from typer.Argument — only local files the user already has access to; remote path uses os.path.basename() to prevent path traversal |
| T-02-07 | Spoofing | infractl/ssh.py | accept | known_hosts=None skips host verification — acceptable for internal tooling where hosts are in ~/.ssh/config; document tradeoff |
| T-02-08 | Denial of Service | infractl/ssh.py | mitigate | asyncio.gather with return_exceptions=True prevents one failed host from blocking others; individual connection timeout via asyncssh defaults |
</threat_model>
