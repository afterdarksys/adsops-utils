def format_output(msg: Message, fmt: str = "text") -> str | bytes:
    if fmt == "json":
        return MessageToJson(msg)
    elif fmt == "proto":
        return msg.SerializeToString()
    else:
        return _proto_to_text(msg)

def print_proto(msg: Message, fmt: str = "text") -> None:
    if fmt == "proto":
        sys.stdout.buffer.write(msg.SerializeToString())
    else:
        print(format_output(msg, fmt))

    for msg in msgs:
        print_proto(msg, fmt)

For `probe_host`:
```python
import asyncio
import asyncssh

async def _probe(hostname: str) -> dict:
    try:
        async with asyncssh.connect(hostname, known_hosts=None) as conn:
            result = await conn.run("echo ok", check=False)
            return {"host": hostname, "reachable": True, "error": None}
    except Exception as e:
        return {"host": hostname, "reachable": False, "error": str(e)}

def probe_host(hostname: str) -> dict:
    return asyncio.run(_probe(hostname))
```
