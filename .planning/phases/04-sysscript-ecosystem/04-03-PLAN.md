    ```python
    load("../../lib/host.star", "host_info")

    base_url = sys.config.get("changes_api_url")
    resp = sys.net.http_get(base_url + "/metrics")
    body = resp["body"]

    request_count = None
    latency_sum = None
    latency_count = None

    for line in body.split("\n"):
        if line.startswith("#") or line == "":
            continue
        if line.startswith("http_requests_total"):
            parts = line.split(" ")
            if len(parts) >= 2:
                request_count = parts[-1]
        if line.startswith("http_request_duration_seconds_sum"):
            latency_sum = line.split(" ")[-1]
        if line.startswith("http_request_duration_seconds_count"):
            latency_count = line.split(" ")[-1]

    latency_avg = 0.0
    if latency_sum and latency_count and float(latency_count) > 0:
        latency_avg = float(latency_sum) / float(latency_count)

    info = host_info()
    print("host:", info["hostname"])
    print("request_count:", request_count)
    print("latency_avg_seconds:", latency_avg)
    ```

    ```python
    import pytest
    from pathlib import Path
    from adsops.sysscript.mock import MockSys
    from adsops.sysscript.runner import SysscriptRunner

    SCRIPT = str(Path(__file__).resolve().parents[4] / "sysscripts" / "services" / "changes-api" / "stats.star")

    PROMETHEUS_BODY = (
        "# HELP http_requests_total The total number of HTTP requests.\n"
        "# TYPE http_requests_total counter\n"
        'http_requests_total{method="GET",code="200"} 1234\n'
        "# HELP http_request_duration_seconds HTTP request duration.\n"
        "# TYPE http_request_duration_seconds histogram\n"
        "http_request_duration_seconds_sum 45.6\n"
        "http_request_duration_seconds_count 1234\n"
    )

    def make_runner(body):
        return SysscriptRunner(sys_global=MockSys({
            "config.get": lambda k: "http://localhost:8080" if k == "changes_api_url" else None,
            "net.http_get": {"status_code": 200, "body": body},
            "exec.run": lambda cmd: (0, "testhost\n", "") if cmd == "hostname" else (0, "Linux\n", ""),
        }))

    def test_parses_request_count():
        result = make_runner(PROMETHEUS_BODY).run(SCRIPT)
        assert result["request_count"] == "1234"

    def test_parses_latency_sum_and_count():
        result = make_runner(PROMETHEUS_BODY).run(SCRIPT)
        assert result["latency_sum"] == "45.6"
        assert result["latency_count"] == "1234"

    def test_computes_latency_avg():
        result = make_runner(PROMETHEUS_BODY).run(SCRIPT)
        assert abs(result["latency_avg"] - (45.6 / 1234)) < 0.001

    def test_empty_body_returns_none():
        result = make_runner("").run(SCRIPT)
        assert result["request_count"] is None

    def test_reads_config_url():
        """Verify script uses sys.config.get('changes_api_url') per D-10."""
        # If config.get returns None for the key, http_get would get None + "/metrics"
        # which would fail. The fact that the test passes proves config.get is called correctly.
        result = make_runner(PROMETHEUS_BODY).run(SCRIPT)
        assert result["request_count"] is not None
    ```

    Note: `exec.run` fixture needed because stats.star loads host.star which calls sys.exec.run.

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| .star script -> sys.net.http_get | Service scripts make HTTP requests to configured URLs |
| .star script -> sys.config.get | Scripts read configuration values (base URLs) |
| load() -> lib .star files | Service scripts load shared helpers from lib/ |

| Threat ID | Category | Component | Disposition | Mitigation Plan |

| T-04-07 | Spoofing | stats.star config.get | mitigate | Base URL comes from sys.config.get (not hardcoded per D-10). In production, agent controls config values. In tests, MockSys provides explicit fixtures. No user-supplied URLs in default execution. |
| T-04-08 | Information Disclosure | stats.star Prometheus parsing | accept | Prometheus metrics are operational data intended for ops consumption. No PII in standard HTTP metrics. |
| T-04-09 | Tampering | stats.star string parsing | accept | Prometheus text format parsing uses simple string split. Malformed metrics body results in None values, not crashes or code execution. Acceptable for internal ops tooling. |
</threat_model>
